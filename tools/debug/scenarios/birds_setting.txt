# SC-BIRD: Settings → Graphics → Birds applies at once on Home's still clock (birds settle away), and Revert brings
# them back. Guards the still-clock settle in scenery/birds/birds_runtime.gd.
eval expr="app.set_meta('birds_locale_before', preferences.language)"
call node=AppRoot method=set_language args=["es"]
open screen=settings
press text=InterfaceCategory
select text=Language item="English"
press text=GraphicsCategory
assert expr="screens.top()._birds_picker.get_item_text(1) == 'Few'" message="Birds choices refresh after language change"
assert expr="screens.top()._birds_picker.tooltip_text.begins_with('Vultures and storks')" message="Birds tooltip refreshes after language change"
select text=Birds item="Off"
wait until="get_node('/root/AppRoot/HomeScene/Field/Scenery/Birds').get_meta('bird_groups')[1].multimesh.visible_instance_count == 0" timeout_s=10
press text="Revert"
wait until="get_node('/root/AppRoot/HomeScene/Field/Scenery/Birds').get_meta('bird_groups')[1].multimesh.visible_instance_count > 0" timeout_s=10
close
assert expr="screens.is_top('home')" message="settings closed"
eval expr="app.set_language(app.get_meta('birds_locale_before'))"
eval expr="app.remove_meta('birds_locale_before')"
