# The X-Cell helicopter: arm, raise the collective briefly, return Home.
fly aircraft=xcell-60-research
# The helicopter opens its launch review after staged loading completes.
wait until="screens.is_top('pause')" timeout_s=120
press text="Continue"
control name=helicopter_arm
assert expr="flight.session.power.armed" message="helicopter armed after launch review"
sticks throttle=0.6 for_s=0.5
wait until="flight.session.sim.tick > 240" timeout_s=120
sticks release=true
home
assert expr="flight == null" message="helicopter flight ended"
