# The X-Cell helicopter: arm, raise the collective briefly, return Home.
fly aircraft=xcell-60-research
# Rendered staged launches open this automatically; headless launches need an explicit review.
# open_pause is idempotent, so this also preserves a rendered launch's existing review.
call node=AppRoot method=open_pause
assert expr="screens.is_top('pause') and flight.session.sim.paused" message="helicopter launch review"
press text="Continue"
control name=helicopter_arm
assert expr="flight.session.power.armed" message="helicopter armed after launch review"
sticks throttle=0.6 for_s=0.5
wait until="flight.session.sim.tick > 240" timeout_s=120
sticks release=true
home
assert expr="flight == null" message="helicopter flight ended"
