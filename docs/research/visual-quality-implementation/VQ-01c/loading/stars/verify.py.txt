import json
import pathlib
import re
import subprocess

ROOT = pathlib.Path(__file__).resolve().parents[2]
OUT = ROOT / 'docs/research/visual-quality-implementation/VQ-01c/loading/stars'
GODOT = ROOT / '.tools/Godot_v4.7.2-stable_win64_console.exe'
ledger = []


def execute(name, args, native=False):
    cmd = [str(GODOT), '--path', str(ROOT / 'app'), '--audio-driver', 'Dummy']
    cmd += ['--resolution', '64x64', '--position', '4000,4000'] if native else ['--headless']
    cmd += args
    p = subprocess.run(cmd, capture_output=True, text=True, timeout=180)
    log = p.stdout + p.stderr
    (OUT / f'{name}.log').write_text(log, encoding='utf-8')
    clean = p.returncode == 0 and not re.search(r'\b(?:SCRIPT\s+)?ERROR\b|WARNING:|\bFAIL\b', log)
    ledger.append(dict(name=name, command=cmd, returncode=p.returncode, clean=clean))
    (OUT / 'verification.json').write_text(json.dumps(ledger, indent=2) + '\n')
    print(name, 'PASS' if clean else 'FAIL', flush=True)
    if not clean:
        print(log[-4000:], flush=True)


for path in ['res://render/world/stars.gd', 'res://tests/test_world_stars.gd',
             str(ROOT / 'research/performance/stars_build_probe.gd')]:
    execute('parse-' + pathlib.Path(path).stem, ['--check-only', '--script', path])
for test in ['test_world_stars', 'test_world_sky', 'test_world_ephemeris',
             'test_sky_appearance_config', 'test_world_presentation_context',
             'test_time_of_day_render_update', 'test_sky_checkpoint']:
    execute(test, ['--script', f'res://tests/{test}.gd'])
execute('native-world-stars', ['--script', 'res://tests/test_world_stars.gd'], native=True)
